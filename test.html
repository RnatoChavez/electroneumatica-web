🔧 Cómo recibir resultados en Google Sheets

Paso 1: Crea una Google Sheet nueva en sheets.new

Paso 2: Ve a Extensiones → Apps Script

Paso 3: Pega este código y guarda (Ctrl+S):

function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = JSON.parse(e.postData.contents);
  
  // Encabezados (solo primera vez)
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(['Fecha','Nombre','Correo','Código',
      'P1','P2','P3','P4','P5','P6','P7','P8','P9','P10',
      'Puntaje','Nota']);
  }
  
  sheet.appendRow([
    new Date(),
    data.nombre,
    data.email,
    data.codigo,
    data.r1, data.r2, data.r3, data.r4, data.r5,
    data.r6, data.r7, data.r8, data.r9, data.r10,
    data.puntaje,
    data.puntaje
  ]);
  
  return ContentService.createTextOutput(JSON.stringify({ok:true}))
    .setMimeType(ContentService.MimeType.JSON);
}
